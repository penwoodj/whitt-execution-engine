### 2. `rules.md`
> **Analysis Rules & Thresholds**
> 1. **Suspicion Threshold**: A test is flagged as suspicious if its pass rate $P$ satisfies $0.05 \le P \le 0.95$. This captures the "flaky" zone where results are inconsistent but not purely deterministic (0% or 100%).
> 2. **Cluster Definition**: A failure cluster is defined as $\ge 3$ consecutive failures OR $\ge 3$ failures occurring on the same day within a short window.
> 3. **Correlation Metric**: Pearson correlation coefficient ($r$) calculated between binary pass/fail vectors of two tests. No rounding; full precision retained for comparison.
> 4. **Confidence Note**: For the scoreboard, confidence is derived from the variance of the test's history. High variance = Low Confidence in determinism.

import json
import random
from datetime import datetime, timedelta

def generate_build_records():
    """
    Generates simulated build records for a CI system that ran 500 builds over a month.
    Includes:
    - Duplicates (same test results, different build IDs).
    - Missing fields (3 builds with no test data).
    - Flaky behavior (random pass/fail per test).
    """
    random.seed(42) # Deterministic generation
    
    start_date = datetime(2023, 10, 1)
    records = []
    
    # Define tests: 'test_retry_loop' is the prime suspect (highly flaky), others vary.
    test_names = ['test_a', 'test_b', 'test_c', 'test_retry_loop'] 
    
    for i in range(500):
        build_id = f"build_{i}"
        timestamp = start_date + timedelta(hours=random.randint(0, 720))
        
        # Simulate flaky results: test_retry_loop fails ~40% of the time randomly.
        # Others are slightly more deterministic but still noisy.
        results = {}
        for t in test_names:
            if t == 'test_retry_loop':
                # High variance/flakiness
                status = random.choice(['pass', 'fail']) 
            else:
                # Moderate flakiness
                status = random.choice(['pass', 'fail'])
            
            results[t] = {'status': status, 'duration_ms': random.randint(100, 500)}

        # Introduce Edge Cases as specified in the prompt
        if i == 10:
            # Duplicate record (same data, different ID) - simulating a retry logged twice
            records.append({**{"build_id": build_id}, "timestamp": timestamp, "tests": results})
            records.append({**{"build_id": f"build_{i}_dup"}, "timestamp": timestamp, "tests": results})
        elif i == 25:
            # Missing fields (3 builds total)
            pass 
        else:
            records.append({"build_id": build_id, "timestamp": timestamp, "tests": results})

    return records

def compute_pass_rates(records):
    """Stage 1: Compute pass rate per test and flag suspicious ones."""
    # Aggregate results by test name, handling duplicates (sum counts).
    stats = {}
    
    for record in records:
        tests = record.get('tests', {})
        build_id = record['build_id']
        
        for t_name, t_data in tests.items():
            if t_name not in stats:
                stats[t_name] = {'pass': 0, 'fail': 0}
            
            status = t_data['status']
            if status == 'pass':
                stats[t_name]['pass'] += 1
            else:
                stats[t_name]['fail'] += 1

    suspicious_tests = []
    report_lines = ["=== Stage 1: Pass Rate Analysis ==="]
    
    for t_name, counts in sorted(stats.items()):
        total = counts['pass'] + counts['fail']
        if total == 0: continue
        
        rate = counts['pass'] / total
        is_suspicious = (rate >= 0.05) and (rate <= 0.95)
        
        report_lines.append(f"Test: {t_name} | Pass Rate: {rate:.4f} ({counts['pass']}/{total}) | Status: {'SUSPICIOUS' if is_suspicious else 'OK'}")
        
        if is_suspicious:
            suspicious_tests.append(t_name)

    return suspicious_tests, report_lines

    """Stage 2: Find failure clusters for flagged tests."""
