def parse_timestamp(ts_str):
    """Attempts to parse ISO8601 or Unix Epoch."""
    if not ts_str:
        return None
    
    # Try Unix Epoch (int or float string)
    try:
        epoch = int(float(ts_str))
        return datetime.utcfromtimestamp(epoch).isoformat() + "Z"
    except ValueError:
        pass

    # Try ISO8601 formats
    patterns = [
        "%Y-%m-%dT%H:%M:%SZ",
        "%Y-%m-%dT%H:%M:%S.%fZ",
        "%Y-%m-%d %H:%M:%S"
    ]
    
    for fmt in patterns:
        try:
            dt = datetime.strptime(ts_str, fmt)
            return dt.isoformat() + "Z"
        except ValueError:
            continue
            
    return None

def group_alerts(alerts):
    """Groups alerts into incidents based on template and time proximity."""
    
    # Sort by parsed timestamp to ensure deterministic grouping order
    sorted_alerts = sorted(alerts, key=lambda x: x["parsed_ts"] or datetime.min.isoformat())
    
    incidents = []
    current_incident = None
    
    # Threshold for noise makers (Rule 2)
    NOISE_THRESHOLD = 10
    
    for alert in sorted_alerts:
        template = alert["template"]
        
        if not current_incident or current_incident["template"] != template:
            # Start new incident
            current_incident = {
                "id": f"INC-{len(incidents)+1}",
                "template": template,
                "alerts": [],
                "start_time": alert["parsed_ts"],
                "end_time": None,
                "service_blast_radius": set(),
                "is_noise_maker": False
            }
        else:
            # Add to existing incident (assuming same template implies same incident for this simplified model)
            current_incident["alerts"].append(alert)
        
        if alert["parsed_ts"]:
            current_incident["end_time"] = alert["parsed_ts"]
            
            # Track services
            if alert["service"]:
                current_incident["service_blast_radius"].add(alert["service"])

    incidents.append(current_incident)
    
    # Identify Noise Makers (Rule 2)
    for inc in incidents:
        if len(inc["alerts"]) > NOISE_THRESHOLD:
            inc["is_noise_maker"] = True
            
    return incidents
