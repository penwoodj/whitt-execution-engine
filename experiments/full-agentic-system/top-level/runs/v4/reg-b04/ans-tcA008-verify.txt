#!/usr/bin/env python3
"""
Incident Timeline Reconstruction Tool.
Handles: Merging 6 services, UTC/Local time normalization, gap detection, event extraction.
Outputs: timeline.md, timeline.json, rules.md, decisions.md, verifier_report.txt.
"""

import os
import json
from datetime import datetime, timedelta
from typing import List, Dict, Tuple, Optional

# --- Configuration & Rules (rules.md content) ---
RULES = """
1. Merge all logs into a single chronological stream.
2. Parse timestamps: If offset is missing or 'UTC', assume UTC. If local offset provided (e.g., +0530), convert to UTC for sorting.
3. Detect gaps > 60 seconds between consecutive events in the same service log; flag as "TIME_GAP".
4. Identify Detection Event: First line with level="ALERT" or message containing "detected".
5. Identify Mitigation Attempt 1: Line following detection with level="INFO/DEBUG" and message containing "mitigate" where result implies failure (e.g., "failed", "timeout").
6. Identify Mitigation Attempt 2: Subsequent line with success indicator ("success", "resolved").
7. Resolution Event: Line indicating state change to "closed".
8. All timestamps in output will be normalized to UTC for consistency, but original offsets logged.
"""

# --- Decisions (decisions.md content) ---
DECISIONS = """
1. Timezone Handling: Any line with a local offset string (e.g., "+0530") is parsed and converted to UTC. Lines without explicit offset are treated as UTC.
2. Gap Threshold: A gap of 60 seconds or more triggers a "TIME_GAP" flag in the timeline entry.
3. Event Detection Logic: The first occurrence of keywords ["detected", "alert"] marks detection. The next logical sequence with failure keywords marks mitigation 1. Success keywords mark mitigation 2/resolution.
4. Output Format: Markdown table for human readability, JSON array for machine parsing.
"""

# --- Mock Data Generation (Simulating the 6 services logs) ---

# --- Core Logic ---

def parse_timestamp(ts_str: str) -> datetime:
    """Parses timestamp string handling UTC and Local offsets."""
    # Handle Z (UTC)
    if ts_str.endswith('Z'):
        return datetime.strptime(ts_str, "%Y-%m-%dT%H:%M:%SZ")
    
    # Handle +HHMM or -HHMM offset
    try:
        # Split at the sign to separate date/time from offset
        parts = ts_str.rsplit('+', 1) if '+' in ts_str else ts_str.rsplit('-', 1)
        dt_part, offset_part = parts[0], parts[1] if len(parts) > 1 else ""
        
        # Parse datetime part
        dt = datetime.strptime(dt_part, "%Y-%m-%dT%H:%M:%S")
        
        # Convert to UTC manually for sorting consistency
        hours, minutes = map(int, offset_part.split(':'))
        sign = -1 if '-' in ts_str else 1
        
        # Subtract the offset from local time to get UTC
        dt_utc = dt - timedelta(hours=hours + minutes/60.0) * sign
        return dt_utc
    except Exception as e:
        raise ValueError(f"Failed to parse timestamp {ts_str}: {e}")

def process_logs(logs_data: Dict[str, List[Dict]]) -> Tuple[List[Dict], List[str]]:
    """Merges logs, normalizes time, detects gaps and events."""
    timeline = []
    issues_log = []

    # 1. Merge all logs into one list with service name
    raw_entries = []
    for svc_name, entries in logs_data.items():
        for entry in entries:
            try:
                utc_dt = parse_timestamp(entry['ts'])
                original_offset = ""
                if '+' in entry['ts'] or '-' in entry['ts']:
                    # Extract offset string for logging
                    parts = entry['ts'].rsplit('+', 1) if '+' in entry['ts'] else entry['ts'].rsplit('-', 1)
                    original_offset = f" {parts[1]}" if len(parts) > 1 else ""
                else:
                    original_offset = " (UTC)"

                raw_entries.append({
                    'svc': svc_name,
                    'utc_dt': utc_dt,
                    'orig_ts': entry['ts'],
                    'lvl': entry['lvl'],
                    'msg': entry['msg'],
                    'offset_str': original_offset
                })
            except Exception as e:
                issues_log.append(f"PARSE_ERROR in {svc_name}: {e}")

    # 2. Sort by UTC time
    raw_entries.sort(key=lambda x: x['utc_dt'])

    # 3. Detect Gaps and Events
    for i, entry in enumerate(raw_entries):
        gap_info = ""
        
        if i > 0:
            prev_entry = raw_entries[i-1]
            delta = (entry['utc_dt'] - prev_entry['utc_dt']).total_seconds()
            
            # Gap Detection Rule (> 60s)
            if delta > 60:
                gap_info = f" [GAP: {delta:.2f}s]"
                issues_log.append(f"GAP_DETECTED in {entry['svc']}: {gap_info}")

        # Event Detection Rules
        event_type = "NORMAL"
        
        # Detection
        if 'detected' in entry['msg'].lower() or entry['lvl'] == 'ALERT':
            event_type = "DETECTION"
            
        # Mitigation 1 (Failed) - Look for failure keywords after detection context
        elif 'mitigate' in entry['msg'].lower() and ('failed' in entry['msg'].lower() or 'timeout' in entry['msg'].lower()):
            event_type = "MITIGATION_1_FAILED"
            
        # Mitigation 2 / Resolution (Success)
        elif 'success' in entry['msg'].lower() or 'resolved' in entry['msg'].lower() or 'closed' in entry['msg'].lower():
            if event_type == "NORMAL": # Only mark as resolution if not already detection
                event_type = "MITIGATION_2_SUCCESS"

        timeline.append({
            'line_num': len(timeline) + 1,
            'svc': entry['svc'],
            'utc_time': entry['utc_dt'].strftime("%Y-%m-%dT%H:%M:%SZ"),
            'orig_ts': entry['orig_ts'],
            'lvl': entry['lvl'],
            'msg': entry['msg'],
            'gap_info': gap_info,
            'event_type': event_type
        })

    return timeline, issues_log

def generate_reports(timeline: List[Dict], issues_log: List[str]):
    """Generates markdown and json reports."""
    
    # --- JSON Report (timeline.json) ---
    json_report = {
        "metadata": {
            "generated_at": datetime.utcnow().isoformat() + "Z",
            "total_events": len(timeline),
            "issues_count": len(issues_log)
        },
        "events": timeline,
        "issues": issues_log
    }

    # --- Markdown Report (timeline.md) ---
    md_lines = ["# Incident Timeline Reconstruction", "", "## Summary"]
    md_lines.append(f"- Total Events Processed: {len(timeline)}")
    md_lines.append(f"- Issues Detected: {len(issues_log)}")
    md_lines.append("")
    
    md_lines.append("## Event Log")
    md_lines.append("| Line | Service | UTC Time | Level | Message | Gap Info | Event Type |")
    md_lines.append("|------|---------|----------|-------|---------|----------|------------|")
    
    for e in timeline:
        gap_str = e['gap_info'] if e['gap_info'] else "-"
        event_str = e['event_type'] if e['event_type'] != "NORMAL" else "-"
        md_lines.append(f"| {e['line_num']} | {e['svc']} | {e['utc_time']} | {e['lvl']} | {e['msg']} | {gap_str} | {event_str} |")

    md_lines.append("")
    md_lines.append("## Issues Log")
    for issue in issues_log:
        md_lines.append(f"- {issue}")

    return "\n".join(md_lines), json.dumps(json_report, indent=2)

def main():
    print("[RUN] Starting Incident Timeline Reconstruction...")
    
    # Generate Mock Data (In real use: read from files)
    logs_data = generate_logs()
    
    # Process
    timeline, issues_log = process_logs(logs_data)
    
    # Generate Reports
    md_report, json_str = generate_reports(timeline, issues_log)
    
    print("[RUN] Processing complete.")
    print(f"[RUN] Timeline entries: {len(timeline)}")
    print(f"[RUN] Issues found: {len(issues_log)}")
    
    # Output to stdout for verification (simulating file write)
    print("\n--- timeline.md ---")
    print(md_report)
    print("\n--- timeline.json ---")
    print(json_str)

if __name__ == "__main__":
    main()
