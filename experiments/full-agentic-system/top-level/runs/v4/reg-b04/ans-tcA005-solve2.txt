def parse_timestamp(ts_str):
    """Parses timestamp strings handling common formats."""
    if not ts_str:
        return None
    for fmt in ["%Y-%m-%d %H:%M:%S", "%Y-%m-%dT%H:%M:%S", "%Y-%m-%d"]:
        try:
            return datetime.strptime(ts_str, fmt)
        except ValueError:
            continue
    # Fallback to just date if time is missing or ambiguous
    try:
        return datetime.strptime(ts_str.split()[0], "%Y-%m-%d")
    except:
        pass
    return None

def calculate_percentiles(records):
    """Calculates p50, p90, p99, p999 in milliseconds."""
    if not records:
        return {"p50": 0, "p90": 0, "p99": 0, "p999": 0}
    
    latencies = [r[1] for r in records]
    sorted_lat = sorted(latencies)
    n = len(sorted_lat)
    
    def percentile(p):
        if n == 0: return 0
        # Use nearest rank method or linear interpolation? 
        # Standard numpy-like approach: index = (p/100) * (n-1)
        idx = (p / 100.0) * (n - 1)
        lower = int(idx)
        upper = min(lower + 1, n - 1)
        weight = idx - lower
        return sorted_lat[lower] * (1 - weight) + sorted_lat[upper] * weight

    return {
        "p50": int(round(percentile(50))),
        "p90": int(round(percentile(90))),
        "p99": int(round(percentile(99))),
        "p999": int(round(percentile(99.9)))
    }

def generate_histogram(records, buckets=10):
    """Generates histogram with method visible."""
    if not records:
        return []
    
    latencies = [r[1] for r in records]
    min_val = min(latencies)
    max_val = max(latencies)
    
    # Handle edge case where all values are same
    if min_val == max_val:
        return [{"range": f"{min_val}-{max_val}", "count": len(latencies)}]

    range_size = (max_val - min_val) / buckets
    histogram = []
    
    for i in range(buckets):
        low = min_val + i * range_size
        high = min_val + (i + 1) * range_size
        
        # Count items in [low, high)
        count = sum(1 for l in latencies if low <= l < high)
        
        histogram.append({
            "range": f"{int(low)}-{int(high)}",
            "count": count
        })
    
    return histogram

def analyze_regression(records_before_wed, records_after_wed):
    """Analyzes /checkout regression specifically."""
    if not records_after_wed:
        return {"start_date": None, "percent_change": 0, "worst_hour_pct": 0}

    p99_after = calculate_percentiles(records_after_wed)["p99"]
    
    if p99_before == 0:
        return {"start_date": None, "percent_change": float('inf'), "worst_hour_pct": 0}
        
    percent_change = ((p99_after - p99_before) / p99_before) * 100
    
    # Find worst single hour (simplified: max p99 in any hour of 'after' set)
    # Group by date+hour for simplicity
    hourly_p99 = defaultdict(list)
    for ts, lat in records_after_wed:
        key = ts.strftime("%Y-%m-%d %H")
        hourly_p99[key].append(lat)
    
    worst_hour_pct = 0.0
    if len(hourly_p99) > 1: # Need comparison
        base_hour = list(hourly_p99.keys())[0]
        base_val = calculate_percentiles(hourly_p99[base_hour])["p99"]
        
        worst_ratio = 0
        for key, vals in hourly_p99.items():
            curr_val = calculate_percentiles(vals)["p99"]
            if base_val > 0:
                ratio = (curr_val - base_val) / base_val * 100
                if ratio > worst_ratio:
                    worst_ratio = ratio
        
        worst_hour_pct = worst_ratio

    return {
        "start_date": records_after_wed[0][0].strftime("%Y-%m-%d") if records_after_wed else None, # Approximate start
        "percent_change": round(percent_change, 2),
        "worst_hour_pct": round(worst_hour_pct, 2)
    }
