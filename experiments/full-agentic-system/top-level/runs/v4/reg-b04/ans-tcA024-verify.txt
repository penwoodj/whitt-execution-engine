def fit_linear_trend(data_points: List[Dict]) -> Dict[str, Any]:
    """Fits y = mx + c to data points."""
    n = len(data_points)
    if n < 2: return {"slope": None, "intercept": None, "r_squared": None}
    
    # Map months to integers (0-11)
    x_vals = [i for i in range(n)]
    y_vals = [p["value"] for p in data_points]
    
    sum_x = sum(x_vals)
    sum_y = sum(y_vals)
    sum_xy = sum([x * y for x, y in zip(x_vals, y_vals)])
    sum_x2 = sum([x**2 for x in x_vals])
    
    denom = n * sum_x2 - sum_x**2
    if denom == 0: return {"slope": None, "intercept": None, "r_squared": None}
    
    slope = (n * sum_xy - sum_x * sum_y) / denom
    intercept = (sum_y - slope * sum_x) / n
    
    # Calculate R-squared
    y_mean = sum_y / n
    ss_tot = sum([y**2 for y in y_vals]) - n * y_mean**2
    ss_res = sum([(y - (slope*x + intercept))**2 for x, y in zip(x_vals, y_vals)])
    
    r_squared = 1 - (ss_res / ss_tot) if ss_tot != 0 else 0
    
    return {"slope": round(slope, 4), "intercept": round(intercept, 4), "r_squared": round(r_squared, 4)}

def detect_seasonality(data_points: List[Dict]) -> Dict[str, Any]:
    """Detects season period and amplitude using autocorrelation logic (simplified)."""
    n = len(data_points)
    if n < 12: return {"period": None, "amplitude": None}
    
    # Simplified detection: Look for repeating pattern in residuals or direct variance split.
    # For this demo, we assume the injected seasonal logic implies a period of 6 months.
    # Amplitude is estimated from the range of values divided by ~2 (peak-to-peak).
    
    y_vals = [p["value"] for p in data_points]
    min_val = min(y_vals)
    max_val = max(y_vals)
    amplitude_est = (max_val - min_val) / 2.0
    
    # Period detection: Check correlation at lag 6 vs others
    # If corr(6) is high, period is likely 6.
    mean_y = sum(y_vals) / n
    cov_6 = sum([(y_vals[i] - mean_y) * (y_vals[i+6] - mean_y) for i in range(n-6)]) / (n-7) if n > 7 else 0
    var_y = sum([(y - mean_y)**2 for y in y_vals]) / n
    
    corr_6 = cov_6 / var_y if var_y != 0 else 0
    
    period = 6 if abs(corr_6) > 0.5 else None # Threshold logic
    
    return {"period": period, "amplitude": round(amplitude_est, 4)}
