### 2. Source Code Files

#### `clean_data.py` (Handles Missing Data & Outliers)
```python
import csv
import sys
from datetime import datetime

INPUT_FILE = "raw_traffic.csv"
CLEANED_FILE = "cleaned_traffic.csv"
LOG_FILE = "cleaning_log.txt"

def log(msg):
    timestamp = datetime.now().isoformat()
    with open(LOG_FILE, 'a') as f:
        f.write(f"[{timestamp}] {msg}\n")

log("Starting data cleaning...")

# Load raw data
raw_data = {}
with open(INPUT_FILE) as f:
    reader = csv.DictReader(f)
    for row in reader:
        month = row['month']
        service = row['service_a'] # Placeholder key, will be replaced by logic below
        
        # Map generic keys to actual services based on column index or name if available
        # Since CSV header is dynamic, we parse values directly.
        pass

# Re-parse strictly by columns assuming standard order: Month, A, B, C, D, E, F
with open(INPUT_FILE) as f:
    reader = csv.reader(f)
    next(reader) # Skip header
    for row in reader:
        month = row[0]
        try:
            val_a = float(row[1]) if row[1].strip() else None
            val_b = float(row[2]) if row[2].strip() else None
            val_c = float(row[3]) if row[3].strip() else None
            val_d = float(row[4]) if row[4].strip() else None
            val_e = float(row[5]) if row[5].strip() else None
            val_f = float(row[6]) if row[6].strip() else None
            
            # Store as dict of lists for processing
            raw_data.setdefault(month, {})
            raw_data[month]['a'] = val_a
            raw_data[month]['b'] = val_b
            raw_data[month]['c'] = val_c
            raw_data[month]['d'] = val_d
            raw_data[month]['e'] = val_e
            raw_data[month]['f'] = val_f
            
        except ValueError:
            log(f"ERROR: Invalid numeric value in row {row}")

# Process Services A, C, E (Linear) and B, D, F (Seasonal/Linear mix based on prompt logic)
# Prompt says: "growth is roughly linear for four services, seasonal for two". 
# We assume A,C,D,E are Linear; B,F are Seasonal. Or vice versa. 
# Let's detect seasonality by checking variance in residuals later. For now, treat all as Linear with optional seasonal adjustment if detected.
# Actually, prompt says: "fit linear trend per service... for two seasonal ones, detect season period".
# We will fit ALL to Linear first, then check B and F for Seasonal patterns (sinusoidal).

services = ['a', 'b', 'c', 'd', 'e', 'f']
cleaned_data = {s: [] for s in services}
log("Processing missing values...")

for month in sorted(raw_data.keys()):
    row = raw_data[month]
    for svc in services:
        val = row.get(svc)
        
        if val is None:
            # Missing Data Rule: Impute with previous valid value (Forward Fill) or 0 if start.
            # Log explicitly.
            log(f"MISSING: {svc} at {month}. Action: Forward fill from last known.")
            # Find last known
            for prev_month in reversed(sorted(raw_data.keys())):
                if raw_data[prev_month].get(svc) is not None:
                    cleaned_data[svc].append(raw_data[prev_month][svc])
                    break
        else:
            cleaned_data[svc].append(val)

# Outlier Rule: December Spike for Service A (500 vs ~155). 
# Detect if value > mean + 3*std. If so, cap at upper bound or flag.
log("Processing outliers...")
for svc in services:
    vals = cleaned_data[svc]
    # Simple outlier detection on the last month (Dec) for A specifically as per prompt "december has obvious outlier spike"
    if svc == 'a' and len(vals) > 10:
        mean_val = sum(vals[:-1]) / (len(vals)-1) # Exclude Dec for baseline
        std_val = ((sum((x-mean_val)**2 for x in vals[:-1]))/(len(vals)-2))**0.5
        
        dec_val = vals[-1]
        if dec_val > mean_val + 3 * std_val:
            log(f"OUTLIER: {svc} at Dec ({dec_val}) exceeds threshold (mean+3std). Action: Cap at mean + 4*std.")
            # Cap it deterministically
            cap = mean_val + 4 * std_val
            cleaned_data[svc][-1] = cap

# Write Cleaned Data
with open(CLEANED_FILE, 'w') as f:
    writer = csv.writer(f)
    header = ['month'] + services
    writer.writerow(header)
    for month in sorted(cleaned_data.keys()):
        row_vals = [month]
        for svc in services:
            row_vals.append(round(cleaned_data[svc][sorted(cleaned_data.keys()).index(month)], 2))
        writer.writerow(row_vals)

log("Data cleaning complete.")
```
