```python
import numpy as np
def hist_10_buckets(data, start=0, width=50):
    # Create bins: [0-50), [50-100)... up to max+width
    min_val = data.min() if data.min() > 0 else 0
    max_val = data.max()
    num_bins = int((max_val - start) / width) + 2
    bins = np.arange(start, max_val + width + 1, width)
    counts, _ = np.histogram(data, bins=bins)
    return counts[:10] # Return top 10 buckets as requested "ten buckets"
```
