import csv
import random
from datetime import timedelta

# Seed for determinism
random.seed(42)

services = ['A', 'B', 'C', 'D', 'E', 'F']
months = list(range(1, 13)) # Jan to Dec

data = []

for service in services:
    base_traffic = random.uniform(800, 1500)
    
    for month in months:
        traffic = base_traffic
        
        # Service A & B: Linear Growth (approx)
        if service == 'A':
            slope = 2.5
            intercept = -30
            trend = slope * month + intercept
            traffic += trend
        elif service == 'B':
            slope = 1.8
            intercept = -20
            trend = slope * month + intercept
            traffic += trend
            
        # Service C & D: Seasonal (Sinusoidal)
        if service == 'C':
            amplitude = 300
            period = 12
            phase = random.uniform(0, 2 * 3.14159)
            seasonal = amplitude * (random.gauss(0, 0.1)) # Simplified seasonality noise
            traffic += seasonal
        elif service == 'D':
            amplitude = 400
            period = 12
            phase = random.uniform(0, 2 * 3.14159)
            seasonal = amplitude * (random.gauss(0, 0.1))
            traffic += seasonal
            
        # Service E & F: Random Walk / Stable
        if service == 'E':
            noise = random.gauss(0, 20)
            traffic += noise
        elif service == 'F':
            noise = random.gauss(0, 30)
            traffic += noise

        # Anomalies: Missing Data (Service A & B in Month 6)
        if service in ['A', 'B'] and month == 6:
            continue
            
        # Anomaly: Outlier Spike (December for everyone)
        if month == 12:
            traffic += random.uniform(500, 800)

data.sort(key=lambda x: (x['service'], x['month']))

with open('raw_data.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['service', 'month', 'traffic'])
    writer.writeheader()
    for row in data:
        writer.writerow(row)

def fit_linear(x_vals, y_vals):
    n = len(x_vals)
    sum_x = sum(x_vals)
    sum_y = sum(y_vals)
    sum_xy = sum([x * y for x, y in zip(x_vals, y_vals)])
    sum_xx = sum([x ** 2 for x in x_vals])
    
    denom = n * sum_xx - sum_x ** 2
    if abs(denom) < 1e-9:
        return 0.0, sum_y / n
    
    slope = (n * sum_xy - sum_x * sum_y) / denom
    intercept = (sum_y - slope * sum_x) / n
    return slope, intercept

def detect_seasonality(x_vals, y_vals):
    # Simple heuristic: check correlation with sin/cos waves for period 12
    best_amp = 0.0
    best_phase = 0.0
    
    for p in [6, 12]:
        sum_sin_sq = sum([math.sin(2 * math.pi * i / p) ** 2 for i in x_vals])
        if sum_sin_sq < 1e-9: continue
        
        # Fit amplitude and phase roughly by minimizing error against sin wave
        best_err = float('inf')
        
        for amp in range(-500, 500, 50):
            for ph in [i * math.pi / 6 for i in range(12)]:
                err = sum([(y - (amp * math.sin(2 * math.pi * x / p + ph))) ** 2 for x, y in zip(x_vals, y_vals)])
                if err < best_err:
                    best_err = err
                    best_amp = amp
                    best_phase = ph
                    
    return best_amp, best_phase
