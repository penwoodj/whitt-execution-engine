### 3. Rules & Decisions (`rules.md`, `decisions.md`)

**`rules.md`**: Explicit logic for handling edge cases.
1.  **Missing Checks Rule**: A check marked as `"missing"` is treated as a data gap, not downtime. It does not count toward the SLO failure calculation or error budget burn.
2.  **SLO Calculation Denominator**: The denominator for SLO compliance is `(Total Minutes Elapsed) - (Minutes with Missing Checks)`.
3.  **Error Budget Burn Rate**: Calculated as `Total Down Minutes / Total Valid Minutes`. If a service has exhausted its budget, the burn rate is effectively infinite or capped at the total available minutes depending on interpretation; here we report the actual ratio of down time to valid time.
4.  **Exhaustion Type Classification**: Determined by input field `exhaustion_type` (e.g., "slow_grind"). If null, default to "unknown".

**`decisions.md`**: Assumptions for unspecified values.
1.  **Sustainable Burn Threshold**: Defined as the average burn rate over the last 3 days if data existed; since we only have Day 9 state, we use the current calculated burn rate as the baseline for comparison (no historical data available).
2.  **Day-Nine Flagging**: Any service where `budget_exhausted` is true triggers a "CRITICAL" flag in output.

### 4. Pipeline Script (`scripts/01_calculate.py`)
This script reads `raw_data/state.json`, applies the rules, and generates `outputs/budget.md` and `outputs/budget.json`. It handles missing checks explicitly.

```python
#!/usr/bin/env python3
import json
import os
from datetime import datetime

# Paths
RAW_DATA_PATH = "raw_data/state.json"
OUTPUT_DIR = "outputs"
LOGS_DIR = "logs"

def load_raw():
    with open(RAW_DATA_PATH) as f:
        return json.load(f)

def calculate_metrics(service):
    total_minutes = service["total_minutes_elapsed"]
    missing_count = service.get("missing_check_count", 0)
    
    # Rule: Missing checks are gaps, not downtime. Exclude from denominator.
    valid_minutes = total_minutes - (missing_count * 1) 
    
    down_minutes = service.get("total_down_minutes", 0)
    
    if valid_minutes > 0:
        burn_rate = down_minutes / valid_minutes
    else:
        burn_rate = float('inf')

    # SLO Compliance Calculation
    # Denominator is valid minutes. Numerator is valid minutes - down minutes.
    compliant_minutes = valid_minutes - down_minutes
    
    if valid_minutes > 0:
        slo_compliance_percent = (compliant_minutes / valid_minutes) * 100
    else:
        slo_compliance_percent = 0.0

    return {
        "service_id": service["id"],
        "slo_target": service["slo_target_percent"],
        "total_minutes": total_minutes,
        "missing_checks_count": missing_count,
        "valid_minutes": valid_minutes,
        "down_minutes": down_minutes,
        "burn_rate": burn_rate,
        "slo_compliance_percent": slo_compliance_percent,
        "budget_exhausted": service.get("budget_exhausted", False),
        "exhaustion_type": service.get("exhaustion_type") or "unknown"
    }

def generate_report(metrics_list):
    md_lines = []
    json_data = {"generated_at": datetime.now().isoformat(), "services": []}
    
    for m in metrics_list:
        # Format burn rate to avoid infinity representation issues if needed, 
        # but keeping raw float for precision as requested.
        burn_str = f"{m['burn_rate']:.6f}" if not isinstance(m['burn_rate'], float) or m['burn_rate'] != float('inf') else "∞"
        
        md_lines.append(f"# {m['service_id']} - Error Budget Analysis")
        md_lines.append("")
        md_lines.append("| Metric | Value |")
        md_lines.append("|---|---|")
        md_lines.append(f"| SLO Target (%) | {m['slo_target']} |")
        md_lines.append(f"| Valid Minutes (Denominator) | {m['valid_minutes']} |")
        md_lines.append(f"| Down Minutes | {m['down_minutes']} |")
        md_lines.append(f"| Burn Rate (Down/Valid) | {burn_str} |")
        md_lines.append(f"| SLO Compliance (%) | {m['slo_compliance_percent']:.6f} |")
        md_lines.append("")
        
        json_data["services"].append({
            "service_id": m["service_id"],
            "slo_target": m["slo_target"],
            "valid_minutes": m["valid_minutes"],
            "down_minutes": m["down_minutes"],
            "burn_rate": m["burn_rate"],
            "slo_compliance_percent": round(m['slo_compliance_percent'], 6), # Keep precision but avoid float noise
            "budget_exhausted": m["budget_exhausted"],
            "exhaustion_type": m["exhaustion_type"]
        })

    with open(os.path.join(OUTPUT_DIR, "budget.md"), "w") as f:
        f.write("\n".join(md_lines))
    
    with open(os.path.join(OUTPUT_DIR, "budget.json"), "w") as f:
        json.dump(json_data, f, indent=2)

def main():
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    raw = load_raw()
    
    metrics_list = []
    for svc in raw["services"]:
        m = calculate_metrics(svc)
        metrics_list.append(m)
        
        # Log specific edge case handling
        if svc.get("missing_check_count", 0) > 0:
            print(f"[LOG] Service {svc['id']}: Handled {svc['missing_check_count']} missing checks by excluding from denominator.")

    generate_report(metrics_list)
    print("[PIPELINE] Report generated successfully.")

if __name__ == "__main__":
    main()
```

### 5. Verification Script (`scripts/02_verify.py`)
This script is independent. It reads `raw_data/state.json` and recomputes the numbers without importing `budget.md` or `budget.json`. If it finds a mismatch, it exits with code 1 and prints an error message.

```python
#!/usr/bin/env python3
import json
import sys
import os

RAW_DATA_PATH = "raw_data/state.json"
OUTPUT_JSON_PATH = "outputs/budget.json"

def verify():
    # Load raw data (Source of Truth)
    with open(RAW_DATA_PATH) as f:
        raw = json.load(f)
    
    # Load generated output
    if not os.path.exists(OUTPUT_JSON_PATH):
        print("[VERIFY] ERROR: Output file does not exist.")
        sys.exit(1)
        
    with open(OUTPUT_JSON_PATH) as f:
        output = json.load(f)

    errors_found = []

    for svc in raw["services"]:
        sid = svc["id"]
        # Find corresponding entry in output
        out_entry = next((e for e in output["services"] if e["service_id"] == sid), None)
        
        if not out_entry:
            errors_found.append(f"[VERIFY] ERROR: Service {sid} missing from output.")
            continue

        total_minutes = svc["total_minutes_elapsed"]
        missing_count = svc.get("missing_check_count", 0)
        valid_minutes_calc = total_minutes - missing_count
        
        down_minutes = svc.get("total_down_minutes", 0)
        
        # Recompute Burn Rate
        if valid_minutes_calc > 0:
            burn_rate_calc = down_minutes / valid_minutes_calc
        else:
            burn_rate_calc = float('inf')

        # Check values against output (allowing for minor float precision differences in JSON serialization)
        out_burn = out_entry["burn_rate"]
        out_slo = out_entry["slo_compliance_percent"]
        
        # Simple equality check with tolerance for floating point noise if not infinity
        if burn_rate_calc != float('inf'):
            if abs(burn_rate_calc - out_burn) > 0.0001:
                errors_found.append(f"[VERIFY] ERROR: Service {sid} Burn Rate mismatch. Expected {burn_rate_calc:.6f}, Got {out_burn}")
        else:
            # If calculated is inf, output should ideally reflect that or be handled specifically
            if out_burn != float('inf') and abs(out_burn - 999999) > 1000: 
                 errors_found.append(f"[VERIFY] ERROR: Service {sid} Burn Rate mismatch (Infinity case). Expected Inf, Got {out_burn}")

        # SLO Compliance Check
        if valid_minutes_calc > 0:
            slo_calc = ((valid_minutes_calc - down_minutes) / valid_minutes_calc) * 100
            if abs(slo_calc - out_slo) > 0.0001:
                errors_found.append(f"[VERIFY] ERROR: Service {sid} SLO Compliance mismatch. Expected {slo_calc:.6f}, Got {out_slo}")

    if errors_found:
        print("[VERIFY] FAILED:")
        for err in errors_found:
            print(err)
        sys.exit(1)
    else:
        print("[VERIFY] PASSED: All numbers recomputed from raw inputs match output.")
        sys.exit(0)

if __name__ == "__main__":
    verify()
```
