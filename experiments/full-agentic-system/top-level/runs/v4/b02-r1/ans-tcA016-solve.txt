#!/usr/bin/env python3
import json
from pathlib import Path

def find_knee(rows):
    knee_points = []
    for i in range(1, len(rows)):
        prev = rows[i-1]
        curr = rows[i]
        
        # Check ordering dependency: Current state vs Previous state
        if (curr['p95'] > prev['p95'] * 1.2) and (curr['throughput'] < prev['throughput']):
            knee_points.append({
                "index": i,
                "users": curr['users'],
                "event": "Latency Spike Detected",
                "action": "Flagged as Knee Point"
            })
            
    # Edge cases: List what was hit (e.g., missing data at knee)
    edge_cases = []
    for point in knee_points:
        if rows[point['index']]['error_rate'] == None:
            edge_cases.append(f"Knee at {point['users']} users had missing error data.")

    return knee_points, edge_cases

def main():
    with open("/tmp/opencode/ramp-run/outputs/ramp.json") as f:
        data = json.load(f)
        
    knees, edges = find_knee(data['rows'])
    
    report = {
        "knees": knees,
        "edge_cases_handled": edges
    }
    
    with open("/tmp/opencode/ramp-run/outputs/knee.md", "w") as f:
        f.write("# Knee Analysis\n")
        for k in knees:
            f.write(f"- **{k['users']} Users**: {k['event']}. Action: {k['action']}\n")

### 4. Pipeline Script: Degradation Verdict (`src/03_degradation.py`)
*Purpose:* Check graceful degradation claim past 600 users. Actual numbers computed, not eyeballed. Traceable to input rows.

```python
#!/usr/bin/env python3
import json
from pathlib import Path

def verify_degradation(rows):
    # Find index where users > 600
    ramp_end_idx = None
    for i, row in enumerate(rows):
        if row['users'] > 600:
            ramp_end_idx = i
            break
            
    if ramp_end_idx is None:
        return {"verdict": "N/A", "reason": "Did not reach 600 users"}

    # Analyze post-600 behavior
    post_600_rows = rows[ramp_end_idx:]
    
    # Check for graceful degradation vs failure
    # Graceful: Errors rise, Throughput drops slowly.
    # Failure: Hard crash (100% error) or erratic spikes without throughput drop correlation.
    
    verdict_details = []
    failed_at = None
    
    for i in range(1, len(post_600_rows)):
        curr = post_600_rows[i]
        prev = post_600_rows[i-1]
        
        # Compute actual numbers from rows
        error_delta = curr['error_rate'] - prev['error_rate'] if curr.get('error_rate') else 0
        
        verdict_details.append({
            "users": curr['users'],
            "throughput_change": curr['throughput'] - prev['throughput'],
            "error_change": error_delta,
            "status": "degrading" if (curr['throughput'] > 0) else "failed"
        })
        
        # If error rate hits critical threshold or throughput drops to zero unexpectedly
        if curr.get('error_rate', 0) >= 5.0: 
             failed_at = curr['users']

    return {
        "verdict": "FAILED" if failed_at else "PASSED",
        "failed_at_users": failed_at,
        "traceable_rows": verdict_details[:5] # Show first few for brevity in log
    }

def main():
    with open("/tmp/opencode/ramp-run/outputs/ramp.json") as f:
        data = json.load(f)
        
    result = verify_degradation(data['rows'])
    
    print(f"[STAGE 3] Degradation Verdict: {result['verdict']}")
    if result['failed_at_users']:
        print(f"System failed gracefully at {result['failed_at_users']} users.")

if __name__ == "__main__":
    main()
```
