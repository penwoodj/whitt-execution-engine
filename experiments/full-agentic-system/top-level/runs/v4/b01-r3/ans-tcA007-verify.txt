def parse_timestamp(ts_str):
    """Attempts to parse timestamp from multiple formats. Returns None if unparseable."""
    if not ts_str:
        return None
    
    # Try primary format first (ISO)
    for fmt in [RULES["timestamp_format_primary"]] + RULES["timestamp_fallback_formats"]:
        try:
            return datetime.strptime(ts_str, fmt)
        except ValueError:
            continue
            
    # Heuristic: If it looks like a date string but failed, maybe it's mixed? 
    # For this pipeline, we reject unparseable timestamps and log them as "Quarantined".
    return None

def group_alerts_into_incidents(alert_list):
    """
    Groups alerts into incidents based on Template and Time Proximity.
    Returns a list of incident dictionaries.
    """
    if not alert_list:
        return []

    # Sort by parsed timestamp to handle out-of-order data logically
    # Alerts with unparseable timestamps are handled separately (Quarantined)

        ts = parse_timestamp(alert.get("timestamp", ""))
        
        if ts is None:

            quarantined_alerts.append({
                "original_index": idx,
                "alert_id": alert.get("incident_id"),
                "reason": "Quarantined: Unparseable Timestamp",
                "disposition": "REJECTED_TIMESTAMP"
            })
            continue
            
        key = (alert.get("template"), ts)
        incidents[key].append(alert)

    # Convert defaultdict to list of structured incidents
    result_incidents = []
    
    for (template, base_time), alerts in incidents.items():
        # Rule: Calculate span and blast radius
        alert_times = [a.get("timestamp", "") for a in alerts]

            "span_minutes": 0, # Calculated below
            "alerts_list": alerts
        }

            times = [parse_timestamp(a.get("timestamp")) for a in alerts]
            if times:

                max_t = max(times)

        result_incidents.append(incident_data)

def identify_noise_makers(incidents):
    """Identifies alerts that fired more than 10 times for the same thing."""

                "template": inc["template"],

                "reason": f"Alert count ({inc['alerts_count']}) exceeds threshold ({RULES['noise_maker_threshold']})"
            })
            
    return noise_makers

def analyze_storm(incident_list):
    """Calculates biggest incident, span, and blast radius."""
    if not incident_list:

    # Sort by alerts_count descending to find the "biggest" one
    sorted_incidents = sorted(incident_list, key=lambda x: x["alerts_count"], reverse=True)
    
    biggest = sorted_incidents[0]

        "biggest_blast_radius_services_count": len(biggest.get("services", [])),
        "total_alerts_processed": sum(i["alerts_count"] for i in incident_list)
    }
