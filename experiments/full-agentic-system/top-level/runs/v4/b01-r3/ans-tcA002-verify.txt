def generate_dataset():

        total = sum(sev_map.values())
        r = random.uniform(0, total)
        cumulative = 0
        for sev, count in sev_map.items():
            cumulative += count
            if r <= cumulative:
                return sev
        return list(sev_map.keys())[0]

def compute_answers():
    """Computes the ground truth answers."""
    
    # Load all data
    def load_jsonl(path):
        data = []
        with open(path, 'r') as f:
            for line in f:
                if line.strip():
                    data.append(json.loads(line))
        return data
    
    critical = load_jsonl(f"{DATA_DIR}/critical.jsonl")
    normal = load_jsonl(f"{DATA_DIR}/normal.jsonl")
    low = load_jsonl(f"{DATA_DIR}/low.jsonl")
    
    all_issues = critical + normal + low
    
    answers = {}
    
    # Q1: Highest resolved count engineer
    resolved_counts = {}
    for issue in all_issues:
        if issue['status'] == 'resolved':
            eng = issue['assignee']
            resolved_counts[eng] = resolved_counts.get(eng, 0) + 1
    
    max_eng = max(resolved_counts, key=resolved_counts.get)
    answers["Q1"] = f"{max_eng} ({resolved_counts[max_eng]})"

    # Q2: % Critical >30 days old and unresolved
    critical_unresolved_old = 0
    total_critical_unresolved = 0
    for issue in all_issues:
        if issue['severity'] == 'critical':
            if issue['status'] != 'resolved':
                total_critical_unresolved += 1
                # Check days_open > 30 (assuming reported_date logic holds)
                try:
                    rep_date = datetime.strptime(issue['reported_date'], "%Y-%m-%d")
                    now = datetime.now()
                    age_days = (now - rep_date).days
                    if age_days > 30:
                        critical_unresolved_old += 1
                except:
                    pass
    
    pct = (critical_unresolved_old / total_critical_unresolved * 100) if total_critical_unresolved else 0
    answers["Q2"] = f"{pct:.2f}%"

    # Q3: Most co-occurring pair
    tag_pairs = {}
    for issue in all_issues:
        tags = issue['tags']
        if len(tags) >= 2:
            for i in range(len(tags)):
                for j in range(i+1, len(tags)):
                    p = tuple(sorted([tags[i], tags[j]]))
                    tag_pairs[p] = tag_pairs.get(p, 0) + 1
    
    best_pair = max(tag_pairs, key=tag_pairs.get) if tag_pairs else ("", "")
    answers["Q3"] = f"{best_pair[0]} and {best_pair[1]}"
