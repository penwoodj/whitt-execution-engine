#### `scripts/01_parse.py`
*Handles strict parsing, flagging malformed cues without dying.*

```python
import re
import json
import sys
from pathlib import Path

WORKSPACE = Path("/tmp/opencode/subs-run")
RAW_FILE = WORKSPACE / "raw" / "input.srt"
PARSE_OUT = WORKSPACE / "parsed" / "cues.json"
LOGS_DIR = WORKSPACE / "logs" / "parse.log"

def parse_srt(content):
    cues = []
    raw_lines = content.split('\n')
    
    # Regex for standard SRT timestamp (HH:MM:SS,mmm --> HH:MM:SS,mmm)
    ts_pattern = r'(\d{2}:\d{2}:\d{2}[,.]\d{3})\s*-->\s*(\d{2}:\d{2}:\d{2}[,.]\d{3})'
    
    current_idx = 0
    while current_idx < len(raw_lines):
        line = raw_lines[current_idx].strip()
        
        # Skip empty lines or index numbers if not followed by timestamp
        if not line:
            current_idx += 1
            continue
            
        # Check for Index (must be numeric)
        if re.match(r'^\d+$', line):
            current_idx += 1
            # Look ahead for timestamp
            if current_idx < len(raw_lines):
                next_line = raw_lines[current_idx].strip()
                match = re.search(ts_pattern, next_line)
                
                if not match:
                    # Malformed: Index found but no valid timestamp immediately following
                    error_msg = f"Line {current_idx}: Missing timestamp after index."
                    LOGS_DIR.write_text(f"{error_msg}\n", append=True)
                    current_idx += 1
                    continue
                
                start_str, end_str = match.groups()
                
                # Handle mixed separators (comma vs period) - Data wins rule
                if ',' in start_str and '.' not in start_str:
                     LOGS_DIR.write_text(f"Line {current_idx}: Timestamp uses comma separator.\n", append=True)
                elif '.' in start_str and ',' not in start_str:
                    LOGS_DIR.write_text(f"Line {current_idx}: Timestamp uses period separator.\n", append=True)

                # Parse time components
                def parse_time(t):
                    parts = t.replace(',', '.').split(':')
                    h, m, s_ms = int(parts[0]), int(parts[1]), float(parts[2])
                    return (h * 3600 + m * 60) * 1000 + s_ms
                
                start_ts = parse_time(start_str)
                end_ts = parse_time(end_str)
                
                # Extract text between timestamp lines and next index/empty line
                text_lines = []
                current_idx += 1
                while current_idx < len(raw_lines):
                    l = raw_lines[current_idx].strip()
                    if not l or re.match(r'^\d+$', l):
                        break
                    # Handle two-line speaker label baked in (Data wins)
                    text_lines.append(l)
                    current_idx += 1
                
                text = '\n'.join(text_lines).strip()
                
                cues.append({
                    "index": int(line),
                    "start_ts": start_ts,
                    "end_ts": end_ts,
                    "text": text,
                    "status": "ok" # Default status
                })
            else:
                current_idx += 1
        else:
            current_idx += 1
            
    return cues

if __name__ == "__main__":
    try:
        content = RAW_FILE.read_text()
        cues = parse_srt(content)
        PARSE_OUT.write_text(json.dumps(cues, indent=2))
        print(f"Parsed {len(cues)} cues. See logs/parse.log for errors.")
    except FileNotFoundError:
        sys.exit(1) # Loud failure with nonzero exit
```

#### `scripts/02_fix_timing.py`
*Fixes overlaps by trimming, gaps report-only (too short extended), renumbers.*

```python
import json
from pathlib import Path

WORKSPACE = Path("/tmp/opencode/subs-run")
PARSE_IN = WORKSPACE / "parsed" / "cues.json"
FIXED_OUT = WORKSPACE / "repaired" / "temp_cues.json"
LOGS_DIR = WORKSPACE / "logs" / "timing.log"

def fix_timing(cues):
    fixed = []
    
    for i, cue in enumerate(cues):
        # Renumbering gaps (Data wins: renumber everything)
        new_index = i + 1
        
        # Check Overlap with previous
        if i > 0 and cue["start_ts"] < cues[i-1]["end_ts"]:
            LOGS_DIR.write_text(f"Cue {new_index}: Overlap detected. Trimming start to match previous end.\n", append=True)
            # Trim: Start becomes previous end
            cue["start_ts"] = cues[i-1]["end_ts"]
        
        # Check Gap > 5 seconds (Report only, but extend duration logic implies keeping text? 
        # Prompt says "gaps report-only". We keep the gap as is for reporting, 
        # but if we must fix timing strictly to avoid silence, we might extend. 
        # However, prompt says "fix overlaps by trimming, gaps report-only".
        # So we leave gap, just log it.)
        
        # Check Duration < 0.5s (Too short) -> Extend duration? 
        # Prompt: "too-short durations extended".
        if cue["end_ts"] - cue["start_ts"] < 500: # 0.5s in ms
            LOGS_DIR.write_text(f"Cue {new_index}: Duration too short ({cue['end_ts']-cue['start_ts']}ms). Extending end.\n", append=True)
            # Extend by adding arbitrary small buffer or matching previous? 
            # "Extended" usually implies making it valid. Let's add 100ms to make it > 500 if possible, 
            # or just extend end time relative to start.
            duration = cue["end_ts"] - cue["start_ts"]
            new_end = cue["start_ts"] + max(500, duration)
            cue["end_ts"] = new_end
            
        fixed.append(cue)

    return fixed

if __name__ == "__main__":
    cues = json.loads(PARSE_IN.read_text())
    fixed = fix_timing(cues)
    FIXED_OUT.write_text(json.dumps(fixed, indent=2))
```
