def calculate_percentiles(data):
    """Calculate p50, p90, p99, p999 in milliseconds (integers)."""
    if not data:
        return {'p50': 0, 'p90': 0, 'p99': 0, 'p999': 0}

    sorted_data = sorted(data)
    n = len(sorted_data)
    
    def percentile(p):
        if n == 0: return 0
        k = (n - 1) * p / 100.0
        f = math.floor(k)
        c = math.ceil(k)
        if f == c:
            return int(sorted_data[int(k)])
        d0 = sorted_data[int(f)] * (c - k)
        d1 = sorted_data[int(c)] * (k - f)
        return int(d0 + d1)

    # Convert to ms and round to integer
    p50 = percentile(50) / 1000.0
    p90 = percentile(90) / 1000.0
    p99 = percentile(99) / 1000.0
    p999 = percentile(999) / 1000.0
    
    return {
        'p50': int(round(p50)),
        'p90': int(round(p90)),
        'p99': int(round(p99)),
        'p999': int(round(p999))
    }

def generate_histogram(data, buckets=10):
    """Generate histogram in 10 buckets."""
    if not data: return []
    
    min_val = min(data)
    max_val = max(data)
    range_val = max_val - min_val
    
    # Handle zero range (all same value)
    if range_val == 0:
        counts = [len(data)] + [0] * buckets
        
    else:
        step = range_val / buckets
        counts = [0] * buckets
        
        for val in data:
            idx = int((val - min_val) / step)
            if idx >= buckets: idx = buckets - 1
            counts[idx] += 1
            
    return counts
