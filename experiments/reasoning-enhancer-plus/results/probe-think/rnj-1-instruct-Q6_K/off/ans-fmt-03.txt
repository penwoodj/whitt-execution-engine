```python
import json

# Given data
total_items = 233
items_per_page = 10
current_page = 4

# Calculate total_pages (counting partial last page)
total_pages = (total_items + items_per_page - 1) // items_per_page

# Determine the number of items on the last page
last_page_items = total_items % items_per_page if total_items % items_per_page != 0 else items_per_page

# Create the pagination envelope as a JSON object
pagination_envelope = {
    "page": current_page,
    "per_page": items_per_page,
    "total": total_items,
    "total_pages": total_pages,
    "last_page_items": last_page_items
}

# Output only the JSON object, one line
print(json.dumps(pagination_envelope))
```
