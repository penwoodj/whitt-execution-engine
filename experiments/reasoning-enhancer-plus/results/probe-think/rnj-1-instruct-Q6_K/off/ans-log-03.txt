```python
instances = {
    "instance-a": {"cost": 100, "ram": 4},
    "instance-b": {"cost": 180, "ram": 8},
    "instance-c": {"cost": 190, "ram": 16},
    "instance-d": {"cost": 260, "ram": 32}
}

# Filter instances based on cost and RAM
filtered_instances = {name: details for name, details in instances.items() if details["cost"] <= 200 and details["ram"] >= 8}

# Find the instance with the most memory
if filtered_instances:
    best_instance = max(filtered_instances, key=lambda x: filtered_instances[x]["ram"])
    print(best_instance)
else:
    print("No suitable instance found.")
```
