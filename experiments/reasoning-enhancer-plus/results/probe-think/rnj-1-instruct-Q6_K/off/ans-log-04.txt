```python
# Define a set to store unique users
unique_users = set()

# List of login events (time, user)
login_events = [
    ("09:01", "A"),
    ("09:03", "A"),
    ("09:05", "B"),
    ("09:06", "C"),
    ("09:09", "C"),
    ("09:11", "A")
]

# Iterate through each login event
for _, user in login_events:
    # Add the user to the set of unique users
    unique_users.add(user)

# Calculate the number of distinct users
num_distinct_users = len(unique_users)

# Print the result as specified
print(f"{num_distinct_users} distinct users")
```
