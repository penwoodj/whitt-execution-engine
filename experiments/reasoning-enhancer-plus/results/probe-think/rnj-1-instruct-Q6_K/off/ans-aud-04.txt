```python
# Define the costs for each service
storage_cost = 210
compute_cost = 180
network_cost = 90

# Calculate the total cost across all services
total_cost = storage_cost + compute_cost + network_cost

# Stated total from the report
stated_total = 500

# Check if the stated total matches the actual total
is_error = total_cost != stated_total

# Output the results
print(f"Actual sum: {total_cost} dollars")
if is_error:
    print("The stated total of 500 dollars is an error.")
else:
    print("The stated total of 500 dollars is correct.")
```
