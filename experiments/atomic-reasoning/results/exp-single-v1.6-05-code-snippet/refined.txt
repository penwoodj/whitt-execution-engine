def find_pairs(numbers: list[int], target: int) -> list[tuple[int, int]]:
    """Find all pairs of numbers that sum to the target value."""
    result = []
    seen = set()
    for num in numbers:
        complement = target - num
        if complement in seen:
            # Ensure consistent ordering by always returning (min, max)
            pair = (min(complement, num), max(complement, num))
            # Only add the pair if it's not already in result
            if pair not in result:
                result.append(pair)
        seen.add(num)
    return result
