def find_pairs(numbers: list[int], target: int) -> list[tuple[int, int]]:
    result = []
    seen = set()
    for num in numbers:
        complement = target - num
        if complement in seen:
            # Ensure the pair is unique and consistent in order (smaller number first)
            if complement <= num:
                result.append((complement, num))
        seen.add(num)
    return result
