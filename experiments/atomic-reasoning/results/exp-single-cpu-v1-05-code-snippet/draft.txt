from typing import List, Tuple

def find_pairs(numbers: List[int], target: int) -> List[Tuple[int, int]]:
    seen = set()
    result = []
    for num in numbers:
        complement = target - num
        if complement in seen and (complement, num) not in result and (num, complement) not in result:
            result.append((min(complement, num), max(complement, num)))
        seen.add(num)
    return sorted(result)
